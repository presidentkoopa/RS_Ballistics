// ============================================================================
// RS_Ballistics settings. Prefix rsb_.
//
// Per-effect numbers live in RSBDEFS profiles, not here. Cvars are the player's
// choices: the dial, the style, switches and multipliers. All user cvars -- looks
// only, each player's own, nothing a netgame compares.
//
// THE OWNER'S RULE (2026-09-14): our effects are ON by default, at good quality.
// A changed default does not reach an ini that already holds a value, so
// Options -> Ballistics & Effects -> "All effects on (recommended)" (KEYCONF
// rsb_all_effects_on) sets these and the engine's effect cvars.
//
// Every one must be read by the ZScript (zscript/rsb/settings.zs):
// tools/menu_lint.py fails the build on one that nothing reads.
// ============================================================================

// ---- the dial
user int    rsb_tier               = 3;       // 0 off, 1 plain, 2 normal, 3 heavy (recommended), 4 extreme
user string rsb_style              = "";      // "" default, or a style profile's name

// ---- impacts
user bool   rsb_impacts            = true;
user float  rsb_impact_particles   = 1.0;
user float  rsb_impact_glow        = 1.0;
user bool   rsb_impact_lights      = true;
user float  rsb_impact_light       = 1.0;
user int    rsb_spark_lights       = 0;         // 0 every spark lights (the owner's answer), 1 capped per burst: a few stronger lights (faster)
user bool   rsb_marks              = true;
user float  rsb_mark_life          = 1.0;
user bool   rsb_ricochets          = true;
user float  rsb_impact_volume      = 1.0;

// ---- muzzle flash
user float  rsb_flash_light        = 1.0;
user float  rsb_flash_size         = 1.0;
user bool   rsb_flash_cone         = true;
user float  rsb_flash_cone_density = 1.0;
user bool   rsb_flash_flame        = true;
user float  rsb_flash_flame_size   = 1.0;
user float  rsb_flash_sparks       = 1.0;
user float  rsb_flash_smoke        = 1.0;
user float  rsb_tail_volume        = 1.0;      // gunshot tails: the room answering each shot
user bool   rsb_flash_powderburns  = true;     // a close wall takes a soot ring and powder stipple
user bool   rsb_flash_blastkick    = true;     // the blast knocks dust off walls beside and a ceiling above, throws casings
user bool   rsb_flash_shockwave    = true;     // big guns throw a bubble of bent air (needs Heat shimmer)

// ---- rounds in flight
user bool   rsb_glide              = true;
user string rsb_round_look         = "";      // "" each round's own look, or a roundlook profile for what flies
user float  rsb_wake               = 1.0;
user bool   rsb_whiz               = true;
user float  rsb_whiz_range         = 1.0;
user float  rsb_whiz_volume        = 1.0;
user bool   rsb_trails             = true;      // energy weapon trails: the railgun's core and corkscrew

// ---- casings
user bool   rsb_casings            = true;
user float  rsb_casing_size        = 1.0;
user bool   rsb_casing_hot         = true;
user float  rsb_casing_life        = 1.0;     // 0..1 of the profile's lifetime: fades sooner, never lingers longer (netplay)
user int    rsb_casing_max         = 300;     // most casings in the world at once; past it the oldest fade (looks only)

// ---- flamethrowers
user bool   rsb_flame              = true;
user float  rsb_flame_particles    = 1.0;
user float  rsb_flame_light        = 1.0;
user float  rsb_flame_volume       = 1.0;
user bool   rsb_flame_tube         = true;      // the drawn-line core of a flame profile with `tube`

// ---- recoil (recoil.zs). THE ONE GAMEPLAY CVAR: a server setting, the same rule for every
// player in a netgame (the owner: "gameplay recoil but as a toggle ... for purists").
server bool sv_rsb_recoil          = true;      // a gun's kick turns where its rounds go
server bool sv_rsb_decals          = false;     // our rounds leave vanilla decals too (off: the wall damage is the mark)
server bool sv_rsb_enemy_rounds    = true;      // monsters' hitscans fly as RS_Ballistics rounds (enemyfire.zs)
server float sv_rsb_enemy_round_speed = 80.0;   // map units a tic
user bool   rsb_enemy_dress        = true;      // with the rounds off: instant enemy shots get our tracer light, crack and hits
user float  rsb_enemy_fx_range     = 2048.0;    // past half: no lights or bent air; past it: no flight or hit looks
user float  rsb_recoil_view        = 1.0;       // the per-shot jolt the arm and the prop show (looks only)

// ---- what you cannot see (Engine docs/EFFECTS_OPTIMIZATION_PLAN.md, M1 and M2): looks only, this machine's view
user bool   rsb_view_cull          = true;      // effects behind your view skip their short-lived sparks, dust, small lights and marks
user float  rsb_lod_range          = 1536.0;    // past this many map units, hits, flashes and wakes draw lighter (0 = never)

// ---- A ROOM ON FIRE (zscript/rsb/lights.zs). The inverse of the rows below, and the opposite KIND of
// thing: where the flame lands is already client-side, so this is presentation and yours alone (user),
// while a shot-out lamp is level state (server). It writes no sector light and nothing the game reads.
user bool  rsb_firelight        = true;    // fire lights the room it burns in (needs a lighting mod listening)
user float rsb_firelight_linger = 3.5;     // seconds a room keeps its glow after the last flame dies

// ---- SHOOTING THE LIGHTS OUT (zscript/rsb/lights.zs). THE ONLY SETTING HERE THAT CHANGES THE GAME:
// a darkened room is level state -- saved, and darker for everyone in co-op. Server, therefore, not user.
server bool  rsb_shootlights       = true;    // shooting a lamp takes it out and darkens the room it lit
server float rsb_shootlights_share = 0.8;     // SHARE of its own light a room loses when every fixture in it is dead; each break takes its part
server float rsb_shootlights_floor = 48.0;    // how dark a room may get this way (a Darkness preset deepens it further)

// ---- LIQUIDS: what comes OUT of something that carries liquid, and the wet it leaves behind.
// Water and coolant only. Blood and gore are a separate job the owner has held (2026-09-18), and
// nothing here touches them. The wet mark is the engine's fourth surface-damage channel (soot,
// hole depth, heat, WET) -- shipped since #17 and, until now, never once painted by us.
user bool  rsb_liquids        = true;     // liquid effects at all: splashes out of water, jets out of a burst pipe, the wet they leave
user float rsb_liquid_splash  = 1.0;      // how much a hit throws up out of water or a pipe, x this
user float rsb_liquid_wet     = 1.0;      // how wet a hit leaves the surface, x this (0 = no wet marks, the hole still gets punched)

// ---- SLOW MOTION (engine build 13): which clock a look that belongs to YOU runs on.
// The world can crawl while you and your gun stay at full speed, so every effect has to pick a
// clock. Ones in the world (bullets, smoke, debris, brass on the floor) always take the world's.
// These are the ones attached to the gun in your hands, where either answer is defensible.
user bool   rsb_slowmo_flash       = true;      // muzzle flash on the REAL clock: a crisp flash even in slow motion (off = it blooms and hangs in the air)

// ---- preview and diagnostics
user string rsb_preview_flame      = "incinerator";
user string rsb_preview_impact     = "bullet";
user string rsb_preview_flash      = "pistol";
user string rsb_preview_ejecta     = "brass_45";
user bool   rsb_debug_surfaces     = false;
user int    rsb_log                = 3;       // 0 off, 1 errors, 2 warnings, 3 loads and first uses, 4 everything
